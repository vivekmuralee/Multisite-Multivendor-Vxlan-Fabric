===============================================================================
Sample rendered configuration - LEAF01
===============================================================================
This is a plain-text example of what Ansible would render onto leaf01 from
the cumulus_underlay and cumulus_overlay roles' templates. It is for
reference/reading only and is not consumed by any playbook.

Role in fabric : leaf (device_role: leaf, not a border leaf)
Management IP  : 192.168.0.21
Loopback IP    : 10.0.0.11/32
VTEP IP        : 10.0.0.11
Router ASN     : 65011
Fabric MTU     : 9216
Uplinks        : swp1 -> spine1, swp2 -> spine2

-------------------------------------------------------------------------------
/etc/network/interfaces (loopback, uplinks, VXLAN overlay)
-------------------------------------------------------------------------------
# This file is managed by Ansible - do not edit by hand

# VTEP source / BGP router-id loopback
auto lo
iface lo inet static
    address 10.0.0.11/32

# BGP-unnumbered fabric links (no IP addressing needed - eBGP peers over IPv6
# link-local, RFC 5549).
auto swp1
iface swp1 inet unnumbered lo
    mtu 9216
    # peer: spine1

auto swp2
iface swp2 inet unnumbered lo
    mtu 9216
    # peer: spine2

# VXLAN EVPN overlay: bridge, per-VLAN L2 VNIs, L3 VNI, distributed anycast
# gateway SVIs, and the tenant VRF (symmetric IRB).

auto bridge
iface bridge
    bridge-vlan-aware yes
    bridge-ports vni10010 vni10020 vni10030 vni10100
    bridge-vids 10 20 30 100
    bridge-pvid 1

auto vni10010
iface vni10010
    vxlan-id 10010
    bridge-access 10
    vxlan-local-tunnelip 10.0.0.11
    mtu 9216

auto vni10020
iface vni10020
    vxlan-id 10020
    bridge-access 20
    vxlan-local-tunnelip 10.0.0.11
    mtu 9216

auto vni10030
iface vni10030
    vxlan-id 10030
    bridge-access 30
    vxlan-local-tunnelip 10.0.0.11
    mtu 9216

# L3 VNI - carries routed (symmetric IRB) traffic between tenant subnets
auto vni10100
iface vni10100
    vxlan-id 10100
    bridge-access 100
    vxlan-local-tunnelip 10.0.0.11
    mtu 9216

# WEB - VLAN 10 / VNI 10010 - distributed anycast gateway
auto vlan10
iface vlan10
    address-virtual 44:38:39:ff:00:01 10.10.10.1/24
    vlan-raw-device bridge
    vlan-id 10
    vrf TENANT_A

# APP - VLAN 20 / VNI 10020 - distributed anycast gateway
auto vlan20
iface vlan20
    address-virtual 44:38:39:ff:00:01 10.10.20.1/24
    vlan-raw-device bridge
    vlan-id 20
    vrf TENANT_A

# DB - VLAN 30 / VNI 10030 - distributed anycast gateway
auto vlan30
iface vlan30
    address-virtual 44:38:39:ff:00:01 10.10.30.1/24
    vlan-raw-device bridge
    vlan-id 30
    vrf TENANT_A

# L3 VNI SVI - anchors the tenant VRF to vni10100
auto vlan100
iface vlan100
    vlan-raw-device bridge
    vlan-id 100
    vrf TENANT_A

auto TENANT_A
iface TENANT_A
    vrf-table auto

-------------------------------------------------------------------------------
/etc/frr/frr.conf (underlay BGP + EVPN overlay + tenant VRF)
-------------------------------------------------------------------------------
! This file is managed by Ansible - do not edit by hand
frr defaults datacenter
hostname leaf01
log syslog informational
service integrated-vtysh-config
!
router bgp 65011
 bgp router-id 10.0.0.11
 bgp bestpath as-path multipath-relax
 no bgp default ipv4-unicast
 neighbor UNDERLAY peer-group
 neighbor UNDERLAY remote-as external
 neighbor UNDERLAY capability extended-nexthop
 neighbor UNDERLAY bfd
 neighbor swp1 interface peer-group UNDERLAY
 neighbor swp2 interface peer-group UNDERLAY
 !
 address-family ipv4 unicast
  network 10.0.0.11/32
  neighbor UNDERLAY activate
  maximum-paths 2
 exit-address-family
 !
 address-family l2vpn evpn
  neighbor UNDERLAY activate
  advertise-all-vni
  advertise-default-gw
 exit-address-family
!
router bgp 65011 vrf TENANT_A
 bgp router-id 10.0.0.11
 !
 address-family ipv4 unicast
  redistribute connected
 exit-address-family
 !
 address-family l2vpn evpn
  advertise ipv4 unicast
 exit-address-family
!

-------------------------------------------------------------------------------
Notes
-------------------------------------------------------------------------------
- leaf01 is a plain leaf (not in the border_leaf group), so the vrf-context
  BGP block above has no external eBGP neighbor - it only redistributes
  connected tenant subnets into EVPN type-5 routes.
- Compare with a border leaf (leaf05/leaf06 + group_vars/border_leaf.yml)
  to see the extra "neighbor <external_uplink.peer_ip> remote-as 65100"
  line that leaks routes to the external WAN/MPLS edge.
